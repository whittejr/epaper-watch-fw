+            moving = (diff > MOTION_THRESHOLD);

+            // Inactivity Tracking (Sleep Monitor)
+            if (diff < SLEEP_MOTION_THRESHOLD) {
+                inactivity_ticks++;
+            } else {
+                inactivity_ticks = 0;
+            }
+        }
